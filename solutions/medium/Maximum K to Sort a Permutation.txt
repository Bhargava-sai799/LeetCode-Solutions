// Title: Maximum K to Sort a Permutation
            // Difficulty: Medium
            // Language: Python3
            // Link: https://leetcode.com/problems/maximum-k-to-sort-a-permutation/

class Solution:
    def sortPermutation(self, nums: List[int]) -> int:
        k = -1  
        for i, x in enumerate(nums):
            if x != i:  
                k &= x 
                if not k:  
                    return 0
        return max(k, 0)
